function generateEllipticalPegs(widthMm: number, heightMm: number, pegCount: number): { x: number; y: number }[] {
  const rx = widthMm / 2;
  const ry = heightMm / 2;
  const pegs: { x: number; y: number }[] = [];
  for (let i = 0; i < pegCount; i++) {
    const angle = -Math.PI / 2 + (2 * Math.PI * i) / pegCount;
    pegs.push({
      x: rx + rx * Math.cos(angle),
      y: ry + ry * Math.sin(angle),
    });
  }
  return pegs;
}

function generateRegularPolygonPegs(diameterMm: number, sides: number, totalPegs: number): { x: number; y: number }[] {
  const r = diameterMm / 2;
  const vertices: { x: number; y: number }[] = [];
  for (let s = 0; s < sides; s++) {
    const angle = -Math.PI / 2 + (2 * Math.PI * s) / sides;
    vertices.push({
      x: r + r * Math.cos(angle),
      y: r + r * Math.sin(angle),
    });
  }
  const pegs: { x: number; y: number }[] = [];
  const pegsPerSide = Math.max(2, Math.floor(totalPegs / sides));
  for (let s = 0; s < sides; s++) {
    const v1 = vertices[s];
    const v2 = vertices[(s + 1) % sides];
    for (let i = 0; i < pegsPerSide; i++) {
      const t = i / pegsPerSide;
      pegs.push({
        x: v1.x + t * (v2.x - v1.x),
        y: v1.y + t * (v2.y - v1.y),
      });
    }
  }
  return pegs;
}

export function getCurrentLoomConfig(): LoomConfig {
  const shape = loomShape.value;
  const pegDiameterMm = Number(fields.pegDiameter.value) || 2;

  if (shape === 'RECTANGULAR') {
    const w = Number(rectWidthInput.value) || 600;
    const h = Number(rectHeightInput.value) || 400;
    const px = Math.max(2, Math.round(Number(rectPegsXInput.value) || 61));
    const py = Math.max(2, Math.round(Number(rectPegsYInput.value) || 41));
    return {
      shape: 'rectangle',
      widthMm: w,
      heightMm: h,
      pegsX: px,
      pegsY: py,
      pegDiameterMm,
    };
  }

  if (shape === 'ELLIPTICAL') {
    const w = Number(rectWidthInput.value) || 600;
    const h = Number(rectHeightInput.value) || 450;
    const count = Math.max(20, Math.round(Number(fields.pegs.value) || 200));
    return {
      shape: 'custom',
      pegs: generateEllipticalPegs(w, h, count),
      pegDiameterMm,
    };
  }

  if (shape === 'REGULAR_POLYGON') {
    const d = Number(fields.diameter.value) || 600;
    const sides = Math.max(3, Math.min(16, Math.round(Number(polygonSidesInput.value) || 6)));
    const total = Math.max(12, Math.round(Number(polyPegsInput.value) || 240));
    return {
      shape: 'custom',
      pegs: generateRegularPolygonPegs(d, sides, total),
      pegDiameterMm,
    };
  }

  if (shape === 'custom') {
    if (!customPegsList || customPegsList.length < 3) {
      return {
        shape: 'circle',
        diameterMm: Number(fields.diameter.value) || 600,
        pegCount: Math.max(3, Math.round(Number(fields.pegs.value) || 200)),
        pegDiameterMm,
        startAngleDeg: -90,
      };
    }
    return {
      shape: 'custom',
      pegs: customPegsList,
      pegDiameterMm,
    };
  }

  // PadrÃ£o: CÃ­rculo
  return {
    shape: 'circle',
    diameterMm: Number(fields.diameter.value) || 600,
    pegCount: Math.max(3, Math.round(Number(fields.pegs.value) || 200)),
    pegDiameterMm,
    startAngleDeg: -90,
  };
}

function updateLoomPreview(): void {
  try {
    const cfg = getCurrentLoomConfig();
    const loom = createLoom(cfg);
    loomRenderer.setLoom(loom);
    miniLoomSummary.textContent = describeLoom(cfg);

    const ctx = miniLoomCanvas.getContext('2d');
    if (!ctx) return;
    const W = miniLoomCanvas.width;
    const margin = 5;
    const extent = loom.extentMm;
    const k = (W - 2 * margin) / extent;
    const X = (mm: number): number => margin + mm * k;

    ctx.clearRect(0, 0, W, W);

    // Borda do tear
    ctx.fillStyle = '#1e1e1e';
    ctx.strokeStyle = '#444';
    ctx.lineWidth = 1;

    if (loom.config.shape === 'circle') {
      const cx = loom.extentMm / 2;
      const cy = loom.extentMm / 2;
      ctx.beginPath();
      ctx.arc(X(cx), X(cy), loom.radiusMm * k, 0, Math.PI * 2);
      ctx.fill();
      ctx.stroke();
    } else if (loom.outline && loom.outline.length > 2) {
      ctx.beginPath();
      loom.outline.forEach((pt, i) => {
        if (i === 0) ctx.moveTo(X(pt.x), X(pt.y));
        else ctx.lineTo(X(pt.x), X(pt.y));
      });
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
    }

    // Pinos
    ctx.fillStyle = '#ff5a36';
    for (const p of loom.pegs) {
      ctx.fillRect(X(p.x) - 0.75, X(p.y) - 0.75, 1.5, 1.5);
    }

    // Atualiza tambÃ©m a visualizaÃ§Ã£o CAM
    blueprintView.setLoomConfig(cfg, Number(fields.pegDiameter.value));
  } catch {
    miniLoomSummary.textContent = 'ConfiguraÃ§Ã£o incompleta.';
  }
}

function syncUIWithConfig(cfg: Readonly<WeaverConfig>, isDirty: boolean): void {
  // Sync shape
  loomShape.value = cfg.shape;
  document.body.dataset.loomShape = cfg.shape;

  controlDiameterMm.hidden = cfg.shape !== 'CIRCULAR';
  controlPolygonSides.hidden = cfg.shape !== 'REGULAR_POLYGON';
  controlWidthMm.hidden = cfg.shape !== 'RECTANGULAR' && cfg.shape !== 'ELLIPTICAL';
  controlCustomPegs.hidden = loomShape.value !== 'custom';

  fields.pegs.value = String(cfg.pinCount);
  fields.diameter.value = cfg.diameterMm.toFixed(1);
  fields.pegDiameter.value = String(cfg.pinDiameterMm);
  fields.threadWidth.value = String(cfg.threadThicknessMm);

  // Sync mode & models
  rasterizerSelect.value = cfg.rasterizer;
  paletteModeSelect.value = cfg.paletteMode;
  controlUserCustomPalette.hidden = cfg.paletteMode !== 'USER_CUSTOM';
  opticalMixModelSelect.value = cfg.opticalMixModel;
  errorMetricSelect.value = cfg.errorMetric;

  // Sync solver & spool
  solverSelect.value = cfg.solver;
  colorStrategy.value = cfg.spoolStrategy;
  fields.lines.value = String(cfg.maxLines);
  fields.repeats.value = String(cfg.maxChordRepeats);

  // Sync algorithm contextual visibility
  controlSpoolStrategy.hidden = cfg.paletteMode === 'MONOCHROME';
  controlBeamWidth.hidden = cfg.solver !== 'BEAM_SEARCH';
  controlFistaIterations.hidden = cfg.solver !== 'GLOBAL_NNLS_FISTA';
  controlSwitchPenaltyLambda.hidden = cfg.paletteMode === 'MONOCHROME' || cfg.spoolStrategy !== 'HYSTERESIS_PENALTY';
  controlMinRunLength.hidden =
    cfg.paletteMode === 'MONOCHROME' ||
    (cfg.spoolStrategy !== 'MIN_RUN_LENGTH' && cfg.spoolStrategy !== 'HYSTERESIS_PENALTY');

  beamWidthInput.value = String(cfg.beamWidth);
  beamDepthInput.value = String(cfg.beamDepth);
  fistaIterationsInput.value = String(cfg.fistaIterations);
  l1RegularizationLambdaInput.value = String(cfg.l1RegularizationLambda);
  switchPenaltyLambdaInput.value = String(cfg.switchPenaltyLambda);
  minRunLengthInput.value = String(cfg.minRunLength);

  // Sync DSP
  fields.claheClip.value = String(cfg.claheClipLimit);
  faceImportanceWeightInput.value = String(cfg.faceImportanceWeight);
  edgeDetectorSelect.value = cfg.edgeDetector;
  controlEdgeBlendWeight.hidden = cfg.edgeDetector === 'NONE';
  edgeBlendWeightInput.value = String(cfg.edgeBlendWeight);
  morphologyOpSelect.value = cfg.morphologyOp;

  // Sync Framing
  fields.cropZoom.value = cfg.framing.zoom.toFixed(2);
  fields.cropPanX.value = String(cfg.framing.panXPercent);
  fields.cropPanY.value = String(cfg.framing.panYPercent);
  cropRotationInput.value = String(cfg.framing.rotationDeg);

  // Dirty state reset button
  btnResetPresetDefaults.hidden = !isDirty;

  updateOutputs();
  checkRepeatsHint();
  updateLoomPreview();
}

function computeActiveCrop(): { x: number; y: number; size: number } | undefined {
  if (!image) return undefined;
  const zoom = Math.max(0.5, Number(fields.cropZoom.value) || 1.0);
  const panXFrac = (Number(fields.cropPanX.value) || 0) / 100;
  const panYFrac = (Number(fields.cropPanY.value) || 0) / 100;

  const baseSize = Math.min(image.width, image.height);
  const cropSize = Math.max(16, baseSize / zoom);

  const cx = image.width / 2 + panXFrac * baseSize;
  const cy = image.height / 2 + panYFrac * baseSize;

  let cropX = cx - cropSize / 2;
  let cropY = cy - cropSize / 2;

  cropX = Math.max(0, Math.min(image.width - cropSize, cropX));
  cropY = Math.max(0, Math.min(image.height - cropSize, cropY));

  return { x: cropX, y: cropY, size: cropSize };
}

function paint(canvas: HTMLCanvasElement, size: number, rgba: ArrayBuffer): void {
  if (canvas === simCanvas && loomRenderer.isWebGL2Supported()) {
    loomRenderer.renderScreenPass();
    return;
  }
  if (canvas.width !== size) {
    canvas.width = size;
    canvas.height = size;
  }
  const ctx = canvas.getContext('2d');
  ctx?.putImageData(new ImageData(new Uint8ClampedArray(rgba), size, size), 0, 0);
}

function start60HzTelemetryLoop(): void {
  function tick(): void {
    if (!running) return;

    loomRenderer.renderFrame();

    const mem = engineClient.memory;
    const currentStep = mem.getCurrentStepIndex();
    const totalMeters = mem.getTotalThreadMeters();
    const switches = mem.getTotalSpoolSwitches();
    const rmse = mem.getCurrentRMSE();
    const deltaE = mem.getCurrentDeltaE();
    const ssim = mem.getCurrentSSIM();
    const status = mem.getStatus();

    const elapsedMs = performance.now() - solverStartTime;
    const manualSec = currentStep * 4.5 + (switches + 1) * 90.0;
    const estManual = currentStep > 0 ? formatDuration(manualSec) : 'â€”';

    setStats(
      currentStep,
      totalMeters * 1000,
      rmse > 0 ? rmse : null,
      elapsedMs,
      undefined,
      undefined,
      deltaE > 0 ? deltaE : null,
      ssim > 0 ? ssim : null,
      switches,
      estManual,
    );
    bar.style.width = `${Math.min(100, (currentStep / currentMaxLines) * 100)}%`;

    if (status === 1 /* RUNNING */) {
      animFrameId = requestAnimationFrame(tick);
    }
  }
  animFrameId = requestAnimationFrame(tick);
}

async function finishWasmRun(isCompleted: boolean): Promise<void> {
  if (animFrameId !== null) {
    cancelAnimationFrame(animFrameId);
    animFrameId = null;
  }
  isWasmRunning = false;
  setRunning(false);
  loomRenderer.renderFrame();

  const mem = engineClient.memory;
  const currentStep = mem.getCurrentStepIndex();
  const totalMeters = mem.getTotalThreadMeters();
  const switches = mem.getTotalSpoolSwitches();
  const rmse = mem.getCurrentRMSE();
  const deltaE = mem.getCurrentDeltaE();
  const ssim = mem.getCurrentSSIM();
  const durationMs = performance.now() - solverStartTime;

  let bomLoaded = false;
  try {
    const bomJsonStr = await engineClient.exportCam('bom_json');
    if (typeof bomJsonStr === 'string' && bomJsonStr.includes('bom')) {
      const parsed = JSON.parse(bomJsonStr) as { bom?: BomDataPayload };
      if (parsed.bom) {
        renderSpoolTableFromBom(parsed.bom);
        bomLoaded = true;
      }
    }
  } catch (err) {
    console.warn('Falha ao obter BOM do weaver-cam:', err);
  }

  const rawSteps = mem.readSteps(0, currentStep);
  const options = readOptions();

  const weaveSteps: WeaveStep[] = rawSteps.map((st, i) => ({
    index: i,
    thread: st.spoolId,
    from: st.fromPin,
    to: st.toPin,
    fromSide: (st.departureWinding === 'CCW' ? 1 : -1) as 1 | -1,
    toSide: (st.arrivalWinding === 'CCW' ? 1 : -1) as 1 | -1,
    gain: st.scoreNormalized,
    segmentMm: 500,
    wrapMm: 3.14,
  }));

  const threadMap = new Map<number, { pegs: number[]; lengthMm: number; lines: number }>();
  for (const st of rawSteps) {
    let t = threadMap.get(st.spoolId);
    if (!t) {
      t = { pegs: [st.fromPin], lengthMm: 0, lines: 0 };
      threadMap.set(st.spoolId, t);
    }
    t.pegs.push(st.toPin);
    t.lines++;
  }

  const threadSummaries: ThreadSummary[] = options.threads.map((cfgThread, i) => {
    const tracked = threadMap.get(i);
    const spoolMeters = mem.getSpoolThreadMeters(i);
    return {
      id: cfgThread.id,
      name: cfgThread.name,
      color: cfgThread.color,
      pegs: tracked ? tracked.pegs : [cfgThread.startPeg],
      lengthMm: spoolMeters > 0 ? spoolMeters * 1000 : (tracked?.lines ?? 0) * 500,
      lines: tracked ? tracked.lines : 0,
    };
  });

  result = {
    version: 1,
    options,
    steps: weaveSteps,
    threads: threadSummaries,
    totalLengthMm: totalMeters * 1000,
    rmse,
    metrics: {
      rmse,
      deltaELab: deltaE,
      ssim,
    },
    stopReason: isCompleted ? 'max-lines' : 'aborted',
    durationMs,
  };

  if (!bomLoaded) {
    renderSpoolTable(result);
  }

  setStats(
    currentStep,
    totalMeters * 1000,
    rmse > 0 ? rmse : null,
    durationMs,
    undefined,
    undefined,
    deltaE > 0 ? deltaE : null,
    ssim > 0 ? ssim : null,
    switches,
    sEstTime.textContent,
  );

  bar.style.width = isCompleted ? '100%' : `${Math.min(100, (currentStep / currentMaxLines) * 100)}%`;
  exportButtons.forEach((b) => (b.disabled = false));
  player.load(result);
}

worker.onmessage = (ev: MessageEvent<FromWorker>) => {
  const msg = ev.data;
  switch (msg.type) {
    case 'target':
      studioManager.updateTargetData(msg.rgba, msg.size, msg.size);
      break;
    case 'progress':
      paint(simCanvas, msg.size, msg.rgba);
      studioManager.setCachedSimulation(msg.rgba);
      bar.style.width = `${(msg.lines / currentMaxLines) * 100}%`;
      setStats(msg.lines, msg.lengthMm, null, msg.elapsedMs, msg.currentThread, msg.currentLayer, null, null, 0);
      break;
    case 'done': {
      result = msg.result;
      if (!result) break;

            departureWinding: st.fromSide === 1 ? 'CCW' : 'CW',
            arrivalWinding: st.toSide === 1 ? 'CCW' : 'CW',
            scoreNormalized: 1.0,
          })),
        );
      } else {
        paint(simCanvas, msg.size, msg.rgba);
      }
      studioManager.setCachedSimulation(msg.rgba);
      bar.style.width = '100%';
      const activeResult = result;
      const switches = activeResult.steps.filter((st, i) => i > 0 && st.thread !== activeResult.steps[i - 1].thread).length;
      setStats(
        activeResult.steps.length,
        activeResult.totalLengthMm,
        activeResult.rmse,
        activeResult.durationMs,
        undefined,
        undefined,
        activeResult.metrics?.deltaELab,
        activeResult.metrics?.ssim,
        switches,
      );
      renderSpoolTable(activeResult);
      exportButtons.forEach((b) => (b.disabled = false));
      setRunning(false);
      player.load(activeResult);
      break;
    }
    case 'error':
      setRunning(false);
      showError(new Error(msg.message));
      break;
  }

function download(name: string, data: BlobPart | Uint8Array, type: string): void {
  const blob = data instanceof Blob ? data : new Blob([contentAsBlobPart(data)], { type });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = name;
  a.click();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

function contentAsBlobPart(data: BlobPart | Uint8Array): BlobPart {
  if (data instanceof Uint8Array) {
    const copy = new Uint8Array(data.byteLength);
    copy.set(data);
    return copy;
  }
  return data;
}

  private bindEvents(): void {
    const canvas = this.el.targetCanvas;
    if (typeof canvas.addEventListener !== 'function') return;

    let isDragging = false;
    let startX = 0;
    let startY = 0;
    let startPanX = 0;

    canvas.addEventListener('pointerdown', (e) => {

      const scaleY = canvas.height / Math.max(1, rect.height);

      startPanY = Number(this.el.panYInput.value) || 0;

      const scaleX = canvas.width / Math.max(1, rect.width);
      const scaleY = canvas.height / Math.max(1, rect.height);
      const px = (e.clientX - rect.left) * scaleX;
      const py = (e.clientY - rect.top) * scaleY;

      if (this.brushActive || this.activeStage === 'importance') {
        this.paintBrushAt(px, py);
        return;

      const deltaPanX = (dx / rect.width) * (80 / zoom);
      const deltaPanY = (dy / rect.height) * (80 / zoom);

      const newPanX = Math.round(Math.max(-50, Math.min(50, startPanX + deltaPanX)));
      const newPanY = Math.round(Math.max(-50, Math.min(50, startPanY + deltaPanY)));

      this.el.panXInput.value = String(newPanX);
      this.el.panYInput.value = String(newPanY);
      this.onFramingChange?.();

    const stopDrag = (e: PointerEvent): void => {
      if (isDragging) {
        isDragging = false;
        try {
          canvas.releasePointerCapture(e.pointerId);
        } catch {
          // ignore
        }
        canvas.style.cursor = this.brushActive || this.activeStage === 'importance' ? 'crosshair' : 'grab';
        if (this.brushActive || this.activeStage === 'importance') {

      },

  private paintBrushAt(px: number, py: number): void {
    if (!this.manualImportanceMask) {
      this.manualImportanceMask = new Float32Array(this.maskWidth * this.maskHeight);
    }
    const w = this.maskWidth;
    const h = this.maskHeight;
    const r = this.brushRadius;
    const sign = this.brushMode === 'boost' ? 1.0 : -1.0;
    const weight = this.brushWeight * sign;

    const minX = Math.max(0, Math.floor(px - r));
    const maxX = Math.min(w - 1, Math.ceil(px + r));
    const minY = Math.max(0, Math.floor(py - r));
    const maxY = Math.min(h - 1, Math.ceil(py + r));

    for (let y = minY; y <= maxY; y++) {
      for (let x = minX; x <= maxX; x++) {
        const dist = Math.hypot(x - px, y - py);
        if (dist <= r) {
          const falloff = 1.0 - dist / r;
          const idx = y * w + x;
          const delta = weight * falloff * 0.25;
          this.manualImportanceMask[idx] = Math.max(-10.0, Math.min(10.0, this.manualImportanceMask[idx] + delta));
        }
      }
    }

    this.refreshCurrentStage();

    // Throttled brush dispatch to worker
    if (this.brushApplyDebounceTimer) {
      clearTimeout(this.brushApplyDebounceTimer);
    }
    this.brushApplyDebounceTimer = setTimeout(() => {
      this.dispatchBrushApply();
    }, 60);
  }

  private dispatchBrushApply(): void {
    if (!this.onBrushApplyHandler || !this.manualImportanceMask) return;
    const result = this.onBrushApplyHandler(this.manualImportanceMask);
    if (result && typeof result.then === 'function') {
      result
        .then((newBuf) => {
          if (newBuf instanceof ArrayBuffer) {
            this.updateImportanceBuffer(newBuf);
          }
        })
        .catch((err) => {
          console.warn('Erro ao atualizar pincel no Worker Wasm:', err);
        });
    }
  }

  // --- Fallback Stage Generators ---

  private computeImportanceMapRgba(w: number, h: number): Uint8ClampedArray {
    const rgba = new Uint8ClampedArray(w * h * 4);
    const cx = w / 2;
    const cy = h * 0.45;
    const rx = w * 0.32;
    const ry = h * 0.38;

    for (let y = 0; y < h; y++) {
      for (let x = 0; x < w; x++) {
        const i = (y * w + x) * 4;
        const normDist = Math.hypot((x - cx) / rx, (y - cy) / ry);
        const facialWeight = Math.max(0, 1.0 - normDist);
        const manual = this.manualImportanceMask ? this.manualImportanceMask[y * w + x] : 0;
        const total = Math.max(0.0, Math.min(1.0, (facialWeight + manual) * 0.7));

        if (total < 0.33) {
          rgba[i] = Math.round((total / 0.33) * 180);
          rgba[i + 1] = 0;
          rgba[i + 2] = 0;
        } else if (total < 0.66) {
          rgba[i] = 255;
          rgba[i + 1] = Math.round(((total - 0.33) / 0.33) * 200);
          rgba[i + 2] = 0;
        } else {
          rgba[i] = 255;
          rgba[i + 1] = 255;
          rgba[i + 2] = Math.round(((total - 0.66) / 0.34) * 255);
        }
        rgba[i + 3] = 255;
      }
    }
    return rgba;
  }

  private computeSobelEdgesRgba(w: number, h: number): Uint8ClampedArray {
    const src = this.wasmPreparedRgba ?? this.cachedPreparedRgba;
    if (!src) return new Uint8ClampedArray(w * h * 4);
    const rgba = new Uint8ClampedArray(w * h * 4);

    const getLum = (x: number, y: number): number => {
      const clampedX = Math.max(0, Math.min(w - 1, x));
      const clampedY = Math.max(0, Math.min(h - 1, y));
      const idx = (clampedY * w + clampedX) * 4;
      return 0.299 * src[idx] + 0.587 * src[idx + 1] + 0.114 * src[idx + 2];
    };

    for (let y = 0; y < h; y++) {
      for (let x = 0; x < w; x++) {
        const idx = (y * w + x) * 4;
        const gx =
          -1 * getLum(x - 1, y - 1) +
          1 * getLum(x + 1, y - 1) +
          -2 * getLum(x - 1, y) +
          2 * getLum(x + 1, y) +
          -1 * getLum(x - 1, y + 1) +
          1 * getLum(x + 1, y + 1);

        const gy =
          -1 * getLum(x - 1, y - 1) +
          -2 * getLum(x, y - 1) +
          -1 * getLum(x + 1, y - 1) +
          1 * getLum(x - 1, y + 1) +
          2 * getLum(x, y + 1) +
          1 * getLum(x + 1, y + 1);

        const mag = Math.min(255, Math.hypot(gx, gy) * 1.5);
        rgba[idx] = Math.round(mag * 0.2);
        rgba[idx + 1] = Math.round(mag * 0.8);
        rgba[idx + 2] = Math.round(mag);
        rgba[idx + 3] = 255;
      }
    }
    return rgba;
  }

  private computeRadonSinogramRgba(w: number, h: number): Uint8ClampedArray {
    const src = this.wasmPreparedRgba ?? this.cachedPreparedRgba;
    if (!src) return new Uint8ClampedArray(w * h * 4);
    const rgba = new Uint8ClampedArray(w * h * 4);

    const numAngles = w;
    const numBins = h;
    const cx = w / 2;
    const cy = h / 2;
    const maxR = Math.hypot(cx, cy);

    for (let col = 0; col < numAngles; col++) {
      const theta = (col / numAngles) * Math.PI;
      const cosT = Math.cos(theta);
      const sinT = Math.sin(theta);

      for (let row = 0; row < numBins; row++) {
        const s = ((row - numBins / 2) / (numBins / 2)) * maxR;
        let sum = 0;
        let count = 0;

        for (let t = -maxR; t <= maxR; t += 4) {
          const px = Math.round(cx + s * cosT - t * sinT);
          const py = Math.round(cy + s * sinT + t * cosT);
          if (px >= 0 && px < w && py >= 0 && py < h) {
            const idx = (py * w + px) * 4;
            const dark = 255 - (0.299 * src[idx] + 0.587 * src[idx + 1] + 0.114 * src[idx + 2]);
            sum += dark;
            count++;
          }
        }

        const avg = count > 0 ? Math.min(255, (sum / count) * 2.2) : 0;
        const i = (row * w + col) * 4;
        rgba[i] = Math.round(avg * 0.3);
        rgba[i + 1] = Math.round(avg * 0.7);
        rgba[i + 2] = Math.round(avg);
        rgba[i + 3] = 255;
      }
    }
    return rgba;
  }

  private computeResidualErrorRgba(w: number, h: number): Uint8ClampedArray {
    const tgt = this.wasmPreparedRgba ?? this.cachedPreparedRgba;
    const sim = this.cachedSimRgba;
    const rgba = new Uint8ClampedArray(w * h * 4);
    if (!tgt) return rgba;

    for (let i = 0; i < w * h * 4; i += 4) {
      const tLum = 0.299 * tgt[i] + 0.587 * tgt[i + 1] + 0.114 * tgt[i + 2];
      const sLum = sim ? 0.299 * sim[i] + 0.587 * sim[i + 1] + 0.114 * sim[i + 2] : 255;
      const err = Math.min(255, Math.abs(tLum - sLum) * 1.5);

      rgba[i] = Math.round(err);
      rgba[i + 1] = Math.round(err * 0.3);
      rgba[i + 2] = Math.round(err * 0.1);
      rgba[i + 3] = 255;
    }
    return rgba;
  }
}

  private drawPaperTiling(ctx: CanvasRenderingContext2D, ox: number, oy: number): void {

    const overlapMm = this.options.overlapMm;

    // Auto orientation: landscape if loom is wider than tall
    const isLand = this.loom.widthMm > this.loom.heightMm;
    const paperW = isLand ? Math.max(rawPaper.w, rawPaper.h) : Math.min(rawPaper.w, rawPaper.h);
    const paperH = isLand ? Math.min(rawPaper.w, rawPaper.h) : Math.max(rawPaper.w, rawPaper.h);

    const printH = Math.max(10, paperH - 2 * printerMarginMm);

    const stepX = Math.max(5, printW - overlapMm);
    const stepY = Math.max(5, printH - overlapMm);

    const pitchR = this.loom.radiusMm;
    const outerR = pitchR + halfRim;
    const innerR = Math.max(0, pitchR - halfRim);

    const bbMinX = -outerR;
    const bbMaxX = outerR;
    const bbMinY = -outerR;
    const bbMaxY = outerR;
    const bbW = bbMaxX - bbMinX;
    const bbH = bbMaxY - bbMinY;

    const numCols = bbW <= printW ? 1 : 1 + Math.ceil((bbW - printW) / stepX);
    const numRows = bbH <= printH ? 1 : 1 + Math.ceil((bbH - printH) / stepY);

    const gridTotalW = printW + (numCols - 1) * stepX;
    const gridTotalH = printH + (numRows - 1) * stepY;

    const startX = bbMinX - (gridTotalW - bbW) / 2;

    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';

      const rowLetter = String.fromCharCode(65 + (r % 26));
      for (let c = 0; c < numCols; c++) {
        const sheetCode = `${rowLetter}${c + 1}`;
        const x0 = startX + c * stepX;
        const x1 = x0 + printW;
        const y0 = startY + r * stepY;
        const y1 = y0 + printH;

        // Check pins inside this sheet
        let pinCount = 0;
        for (const p of this.loom.pegs) {
          const wx = p.x - ox;
          const wy = p.y - oy;
          if (wx >= x0 && wx <= x1 && wy >= y0 && wy <= y1) {
            pinCount++;
          }
        }

        // Distance evaluation for culling empty interior/exterior sheets
        const corners = [
          [x0, y0],
          [x1, y0],
          [x0, y1],
          [x1, y1],
        ];
        let dMax = 0;
        for (const [cx, cy] of corners) {
          const d = Math.hypot(cx, cy);
          if (d > dMax) dMax = d;
        }

        const dxMin = 0 < x0 ? x0 : 0 > x1 ? -x1 : 0;
        const dyMin = 0 < y0 ? y0 : 0 > y1 ? -y1 : 0;

        // Empty interior (central hollow) culling
        const isEmptyInterior = innerR > 1.0 && dMax < innerR && pinCount === 0;
        // Empty exterior culling

        const isActive = !isEmptyInterior && !isEmptyExterior;

          ctx.fillStyle = 'rgba(14, 165, 233, 0.05)';
          ctx.fillRect(x0, y0, printW, printH);

          // Outer print boundary
          ctx.strokeStyle = 'rgba(56, 189, 248, 0.45)';
          ctx.lineWidth = 1.0 / this.scale;
          ctx.setLineDash([5 / this.scale, 3 / this.scale]);
          ctx.strokeRect(x0, y0, printW, printH);

              ctx.fillRect(x0, y1 - overlapMm, printW, overlapMm);

export interface PreparedImageResult {

export interface ProgressNotification {
  currentStep: number;
  rmse: number;

/**
 * High-performance client that bridges the UI thread to the background Wasm solver worker.
 */
export class WeaverEngineClient {
  public readonly memory: SharedWorkspaceMemory;
  private worker: Worker | null = null;
  private isInitialized = false;
  private wasmReady = false;

  private onProgressCallbacks: Set<(progress: ProgressNotification) => void> = new Set();
  private onDoneCallbacks: Set<(done: ProgressNotification) => void> = new Set();
  private onPausedCallbacks: Set<(step: number) => void> = new Set();
  private onErrorCallbacks: Set<(err: string) => void> = new Set();

  private pendingCamExports: Map<
    string,
    { resolve: (res: string | Uint8Array) => void; reject: (err: unknown) => void }
  > = new Map();
  private pendingImagePrepare: { resolve: (res: PreparedImageResult) => void; reject: (err: unknown) => void } | null =
    null;
  private pendingSolverInit: { resolve: () => void; reject: (err: unknown) => void } | null = null;
  private pendingBrushApply: { resolve: (buf: ArrayBuffer) => void; reject: (err: unknown) => void } | null = null;

  constructor(options?: EngineClientOptions) {
    this.memory = new SharedWorkspaceMemory(options?.capacitySteps, {
      forceFallback: options?.forceFallback,
    });
  }

  /**
   * Initializes the Web Worker and connects the SharedArrayBuffer memory.
   */
  public async init(workerOverride?: Worker): Promise<void> {
    if (this.isInitialized) return;

    return new Promise((resolve, reject) => {
      try {
        if (workerOverride) {
          this.worker = workerOverride;
        } else if (typeof Worker !== 'undefined') {
          this.worker = new Worker(new URL('./solver.worker.ts', import.meta.url), {
            type: 'module',
          });
        } else {
          // Ambiente sem Worker (ex: testes em Node)
          this.isInitialized = true;
          this.wasmReady = false;
          resolve();
          return;
        }

        const timeout = setTimeout(() => {
          reject(
            new Error(
              "Timeout ao inicializar o Wasm solver worker. Verifique se o mÃ³dulo Wasm foi compilado via 'npm run build:wasm'.",
            ),
          );
        }, 8000);

        this.worker.onmessage = (e: MessageEvent<FromSolverWorkerMessage>) => {
          const msg = e.data;
          switch (msg.type) {
            case 'READY':
              clearTimeout(timeout);
              this.isInitialized = true;
              this.wasmReady = true;
              resolve();
              break;

            case 'IMAGE_PREPARED':
              if (this.pendingImagePrepare) {
                this.pendingImagePrepare.resolve({
                  summaryJson: msg.summaryJson,
                  preparedRgba: msg.preparedRgba,
                  importanceRgba: msg.importanceRgba,
                  edgesRgba: msg.edgesRgba,
                  radonRgba: msg.radonRgba,
                  dimensions: msg.dimensions,
                });
                this.pendingImagePrepare = null;
              }
              break;

            case 'BRUSH_APPLIED':
              if (this.pendingBrushApply) {
                this.pendingBrushApply.resolve(msg.importanceRgba);
                this.pendingBrushApply = null;
              }
              break;

            case 'SOLVER_INITIALIZED':
              if (this.pendingSolverInit) {
                this.pendingSolverInit.resolve();
                this.pendingSolverInit = null;
              }
              break;

            case 'PROGRESS':
              for (const cb of this.onProgressCallbacks) {
                cb({
                  currentStep: msg.currentStep,
                  rmse: msg.rmse,
                  deltaE: msg.deltaE,
                  ssim: msg.ssim,
                  totalThreadMeters: msg.totalThreadMeters,
                });
              }
              break;

            case 'PAUSED':
              for (const cb of this.onPausedCallbacks) {
                cb(msg.currentStep);
              }
              break;

            case 'DONE':
              for (const cb of this.onDoneCallbacks) {
                cb({
                  currentStep: msg.currentStep,
                  rmse: msg.rmse,
                  deltaE: msg.deltaE,
                  ssim: msg.ssim,
                  totalThreadMeters: msg.totalThreadMeters,
                });
              }
              break;

            case 'CAM_EXPORT_RESULT': {
              const pending = this.pendingCamExports.get(msg.exportId);
              if (pending) {
                pending.resolve(msg.result);
                this.pendingCamExports.delete(msg.exportId);
              }
              break;
            }

            case 'ERROR':
              if (this.pendingImagePrepare) {
                this.pendingImagePrepare.reject(new Error(msg.message));
                this.pendingImagePrepare = null;
              }
              if (this.pendingSolverInit) {
                this.pendingSolverInit.reject(new Error(msg.message));
                this.pendingSolverInit = null;
              }
              for (const cb of this.onErrorCallbacks) {
                cb(msg.message);
              }
              if (msg.fatal) {
                clearTimeout(timeout);
                reject(new Error(msg.message));
              }
              break;
          }
        };

        this.worker.onerror = (err) => {
          clearTimeout(timeout);
          const message = err.message || 'Erro desconhecido no Web Worker';
          for (const cb of this.onErrorCallbacks) {
            cb(message);
          }
          reject(new Error(`Worker error: ${message}`));
        };

        // Dispara mensagem inicial de conexÃ£o
        const initMsg: ToSolverWorkerMessage = {
          type: 'INIT',
          buffer: this.memory.buffer,
          capacitySteps: this.memory.capacitySteps,
        };
        this.worker.postMessage(initMsg);
      } catch (err) {
        reject(
          new Error(
            `Falha ao iniciar WeaverEngineClient: ${err instanceof Error ? err.message : String(err)}. Verifique se 'npm run build:wasm' foi executado.`,
          ),
        );
      }
    });
  }

  /**
   * Executes the complete DSP pipeline on the input RGBA image in the background.
   */
  public async loadAndPrepareImage(
    rawRgba: Uint8Array | ArrayBuffer,
    width: number,
    height: number,
    config: WeaverConfig | string,
  ): Promise<PreparedImageResult> {
    if (!this.worker) {
      throw new Error("Worker nÃ£o inicializado. Chame 'await client.init()' primeiro.");
    }

    const configJson = typeof config === 'string' ? config : JSON.stringify(config);
    let rawBuffer: ArrayBuffer;
    if (rawRgba instanceof Uint8Array) {
      rawBuffer = new ArrayBuffer(rawRgba.byteLength);
      new Uint8Array(rawBuffer).set(rawRgba);
    } else if (rawRgba instanceof ArrayBuffer) {
      rawBuffer = rawRgba;
    } else {
      const u8 = new Uint8Array(rawRgba);
      rawBuffer = new ArrayBuffer(u8.byteLength);
      new Uint8Array(rawBuffer).set(u8);
    }

    return new Promise((resolve, reject) => {
      this.pendingImagePrepare = { resolve, reject };
      const msg: ToSolverWorkerMessage = {
        type: 'LOAD_AND_PREPARE_IMAGE',
        rawRgba: rawBuffer,
        width,
        height,
        configJson,
      };
      this.worker!.postMessage(msg, [rawBuffer]);
    });

async function handlePrepareImage(
  msg: Extract<ToSolverWorkerMessage, { type: 'LOAD_AND_PREPARE_IMAGE' }>,

  if (!wasmEngine) {
    post({ type: 'ERROR', message: 'Wasm engine not initialized' });
    return;
  }

  try {
    const rawBytes = new Uint8Array(msg.rawRgba);
    const summaryJson = wasmEngine.load_and_prepare_image(rawBytes, msg.width, msg.height, msg.configJson);

    const prep = wasmEngine.get_prepared_target_rgba();

    function copyToConventionalArrayBuffer(view: Uint8Array): ArrayBuffer {
      const copy = new ArrayBuffer(view.byteLength);
      new Uint8Array(copy).set(view);
      return copy;
    }

    const prepBuf = copyToConventionalArrayBuffer(prep);
    const impBuf = copyToConventionalArrayBuffer(imp);
    const edgeBuf = copyToConventionalArrayBuffer(edge);
    const radonBuf = copyToConventionalArrayBuffer(radon);

        edgesRgba: edgeBuf,

async function runSolverBatchLoop(batchSize = 50): Promise<void> {
  if (!wasmEngine || !memory) return;

  isRunning = true;
  isPaused = false;
  isAborted = false;
  memory.setStatus(SolverStatusFlag.RUNNING);

  while (isRunning && !isPaused && !isAborted) {
    // 1. Executa lote de passos no nÃºcleo Wasm/Rust
    const stillRunning = wasmEngine.step_solver_batch(batchSize);

    const currentStep = memory.getCurrentStepIndex();
    const rmse = memory.getCurrentRMSE();
    const deltaE = memory.getCurrentDeltaE();
    const ssim = memory.getCurrentSSIM();
    const totalThreadMeters = memory.getTotalThreadMeters();

    // 2. Notifica progresso em tempo real para a Main Thread
    post({
      type: 'PROGRESS',
      currentStep,
      rmse,
      deltaE,
      ssim,
      totalThreadMeters,
    });

    if (!stillRunning) {
      isRunning = false;
      const status = memory.getStatus();
      if (status === SolverStatusFlag.PAUSED) {
        post({ type: 'PAUSED', currentStep });
      } else if (status === SolverStatusFlag.ABORT) {
        post({ type: 'ABORTED', currentStep });
      } else {
        post({
          type: 'DONE',
          currentStep,
          rmse,
          deltaE,
          ssim,
          totalThreadMeters,
        });
      }
      break;
    }

    // 3. Cede a vez na fila de tarefas (event loop) para permitir receber
    // mensagens de PAUSE, ABORT ou APPLY_BRUSH entre os lotes
    await new Promise((resolve) => setTimeout(resolve, 0));
  }
}

function handleStart(msg: Extract<ToSolverWorkerMessage, { type: 'START' }>): void {
  if (isRunning) return;
  const batch = msg.batchSize ?? 50;
  void runSolverBatchLoop(batch);
}

  isPaused = true;
  isRunning = false;
  if (memory) {

}

  isAborted = true;
  isRunning = false;
  if (memory) {
    memory.setStatus(SolverStatusFlag.ABORT);
  }
  const currentStep = memory?.getCurrentStepIndex() ?? 0;
  post({ type: 'ABORTED', currentStep });
}

/// Modelo de mistura Ã³ptica e fÃ­sica de transmitÃ¢ncia.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum OpticalMixModelType {
    /// OclusÃ£o fÃ­sica Porter-Duff com alfa-blending $C^{(t+1)} = (1 - \alpha w)C^{(t)} + (\alpha w)\mathbf{s}_k$.
    PorterDuffOcclusion,
    /// Mistura linear subtrativa em intensidade de pigmento.
    LinearSubtractive,

/// Pesos perceptuais diagonais da mÃ©trica $[w_0, w_1, w_2]$ para cÃ¡lculo rÃ¡pido do produto interno ponderado.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct MetricWeights {
    pub w0: f32,
    pub w1: f32,
    pub w2: f32,
}

impl MetricWeights {
    /// MÃ©trica euclidiana nÃ£o ponderada em Linear RGB $[1.0, 1.0, 1.0]$.
    pub const LINEAR_RGB: Self = Self {
        w0: 1.0,
        w1: 1.0,
        w2: 1.0,
    };

    /// Coeficientes de luminÃ¢ncia Rec. 709.
    pub const REC709_LUMA: Self = Self {
        w0: LUMA_REC709_R,
        w1: LUMA_REC709_G,
        w2: LUMA_REC709_B,
    };

    /// Calcula pesos locais aproximados de Redmean para um pixel alvo e tela atual.
    #[inline]
    pub fn from_redmean(c_current_r: f32, target_r: f32) -> Self {
        let r_mean = (c_current_r + target_r) * 0.5;
        Self {
            w0: 2.0 + r_mean,
            w1: 4.0,
            w2: 3.0 - r_mean,
        }
    }

    /// Deriva os pesos da mÃ©trica a partir do `ColorErrorMetric` selecionado.
    #[inline]
    pub fn for_pixel(
        metric: ColorErrorMetric,
        c_current_r: f32,
        target_r: f32,
    ) -> Self {
        match metric {
            ColorErrorMetric::LinearRgb => Self::LINEAR_RGB,
            ColorErrorMetric::RedmeanWeighted => Self::from_redmean(c_current_r, target_r),
            ColorErrorMetric::CielabDeltaE76 | ColorErrorMetric::CielabDeltaE2000 => {
                // Para ganho marginal instantÃ¢neo no loop interno de milhÃµes de avaliaÃ§Ãµes,
                // a mÃ©trica Rec. 709 / Redmean linearizada provÃª aproximaÃ§Ã£o hiper-rÃ¡pida
                // fortemente correlacionada com L* e croma CIELAB.
                Self::REC709_LUMA
            }
        }
    }

    /// Produto interno ponderado $\langle \mathbf{u}, \mathbf{v} \rangle_{\mathbf{M}}$.
    #[inline]
    pub fn dot(&self, u: [f32; 3], v: [f32; 3]) -> f32 {
        self.w0 * u[0] * v[0] + self.w1 * u[1] * v[1] + self.w2 * u[2] * v[2]
    }

    /// Norma quadrÃ¡tica ponderada $\|\mathbf{u}\|_{\mathbf{M}}^2$.
    #[inline]
    pub fn norm_sq(&self, u: [f32; 3]) -> f32 {
        self.w0 * u[0] * u[0] + self.w1 * u[1] * u[1] + self.w2 * u[2] * u[2]
    }
}

/// Avaliador de modelos Ã³pticos e cÃ¡lculo de ganho marginal.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct OpticalModelEvaluator {
    pub mix_model: OpticalMixModelType,
    pub error_metric: ColorErrorMetric,
}

impl OpticalModelEvaluator {
    pub const fn new(mix_model: OpticalMixModelType, error_metric: ColorErrorMetric) -> Self {
        Self {
            mix_model,
            error_metric,
        }
    }

    /// Calcula a cor atualizada $C^{(t+1)}$ de um pixel $p$ sob a aplicaÃ§Ã£o de um fio de cor $\mathbf{s}_k$
    /// com opacidade efetiva $\alpha_{\text{efetivo}}$ e peso de cobertura sub-pixel $w \in [0.0, 1.0]$.
    #[inline]
    pub fn update_pixel_color(
        &self,
        current: LinearRgbColor,
        spool_color: LinearRgbColor,
        alpha_eff: f32,
        subpixel_weight: f32,
    ) -> LinearRgbColor {
        let aw = (alpha_eff * subpixel_weight).clamp(0.0, 1.0);
        match self.mix_model {
            OpticalMixModelType::PorterDuffOcclusion => {
                // C^{(t+1)} = (1 - \alpha w) C^{(t)} + (\alpha w) s_k
                LinearRgbColor::new(
                    (1.0 - aw) * current.r + aw * spool_color.r,
                    (1.0 - aw) * current.g + aw * spool_color.g,
                    (1.0 - aw) * current.b + aw * spool_color.b,
                )
            }
            OpticalMixModelType::LinearSubtractive => {
                // Subtrativo: absorve luz proporcionalmente a (1 - s_k)
                LinearRgbColor::new(
                    (current.r - aw * (1.0 - spool_color.r)).clamp(0.0, 1.0),
                    (current.g - aw * (1.0 - spool_color.g)).clamp(0.0, 1.0),
                    (current.b - aw * (1.0 - spool_color.b)).clamp(0.0, 1.0),
                )
            }
            OpticalMixModelType::BeerLambertTransmittance => {
                // Multiplicativo: T^{(t+1)} = T^{(t)} * (1 - aw * (1 - s_k))
                LinearRgbColor::new(
                    (current.r * (1.0 - aw * (1.0 - spool_color.r))).clamp(0.0, 1.0),
                    (current.g * (1.0 - aw * (1.0 - spool_color.g))).clamp(0.0, 1.0),
                    (current.b * (1.0 - aw * (1.0 - spool_color.b))).clamp(0.0, 1.0),
                )
            }
        }
    }

    /// Calcula o ganho marginal de reduÃ§Ã£o de erro $\Delta E$ em um Ãºnico pixel $p$:
    ///
    /// Para `PorterDuffOcclusion`:
    /// $$\text{Ganho}(p, k) = W(p) \cdot \left[ 2(\alpha w) \langle \mathbf{e}(p), \mathbf{d}_k(p) \rangle_{\mathbf{M}} - (\alpha w)^2 \|\mathbf{d}_k(p)\|_{\mathbf{M}}^2 \right]$$
    /// onde $\mathbf{e}(p) = C^{(t)}(p) - T(p)$ e $\mathbf{d}_k(p) = C^{(t)}(p) - \mathbf{s}_k$.
    ///
    /// Para os demais modelos ou avaliaÃ§Ã£o perceptual estrita, calcula a diferenÃ§a direta
    /// entre o erro anterior ponderado e o novo erro.
    #[inline]
    pub fn pixel_marginal_gain(
        &self,
        current: [f32; 3],
        target: [f32; 3],
        spool: [f32; 3],
        alpha_eff: f32,
        subpixel_weight: f32,
        importance_w: f32,
    ) -> f32 {
        if importance_w <= 0.0 {
            return 0.0;
        }

        let aw = (alpha_eff * subpixel_weight).clamp(0.0, 1.0);
        if aw <= 1e-7 {
            return 0.0;
        }

        let weights = MetricWeights::for_pixel(self.error_metric, current[0], target[0]);

        match self.mix_model {
            OpticalMixModelType::PorterDuffOcclusion => {
                // e(p) = C^{(t)}(p) - T(p)
                let e = [
                    current[0] - target[0],
                    current[1] - target[1],
                    current[2] - target[2],
                ];
                // d_k(p) = C^{(t)}(p) - s_k
                let d = [
                    current[0] - spool[0],
                    current[1] - spool[1],
                    current[2] - spool[2],
                ];

                let dot_ed = weights.dot(e, d);
                let norm_d_sq = weights.norm_sq(d);

                let raw_gain = 2.0 * aw * dot_ed - (aw * aw) * norm_d_sq;
                importance_w * raw_gain
            }
            OpticalMixModelType::LinearSubtractive => {
                let e_old = [
                    current[0] - target[0],
                    current[1] - target[1],
                    current[2] - target[2],
                ];
                let c_new = [
                    (current[0] - aw * (1.0 - spool[0])).clamp(0.0, 1.0),
                    (current[1] - aw * (1.0 - spool[1])).clamp(0.0, 1.0),
                    (current[2] - aw * (1.0 - spool[2])).clamp(0.0, 1.0),
                ];
                let e_new = [
                    c_new[0] - target[0],
                    c_new[1] - target[1],
                    c_new[2] - target[2],
                ];
                let err_old = weights.norm_sq(e_old);
                let err_new = weights.norm_sq(e_new);
                importance_w * (err_old - err_new)
            }
            OpticalMixModelType::BeerLambertTransmittance => {
                let e_old = [
                    current[0] - target[0],
                    current[1] - target[1],
                    current[2] - target[2],
                ];
                let c_new = [
                    (current[0] * (1.0 - aw * (1.0 - spool[0]))).clamp(0.0, 1.0),
                    (current[1] * (1.0 - aw * (1.0 - spool[1]))).clamp(0.0, 1.0),
                    (current[2] * (1.0 - aw * (1.0 - spool[2]))).clamp(0.0, 1.0),
                ];
                let e_new = [
                    c_new[0] - target[0],
                    c_new[1] - target[1],
                    c_new[2] - target[2],
                ];
                let err_old = weights.norm_sq(e_old);
                let err_new = weights.norm_sq(e_new);
                importance_w * (err_old - err_new)
            }
        }
    }

    /// Calcula o ganho marginal acumulado ao longo de uma corda rasterizada:
    /// $\text{Ganho}_{\text{corda}} = \sum_{p \in \text{corda}} \text{Ganho}(p, k)$.
    ///
    /// Consulta os pixels e pesos sub-pixel diretamente da struct `CanvasStateSoA`.
    pub fn chord_marginal_gain(
        &self,
        state: &CanvasStateSoA,
        pixel_indices: &[u32],
        subpixel_weights_u8: &[u8],
        spool: &SpoolColor,
        alpha_eff: f32,
    ) -> f32 {
        let spool_arr = [spool.color.r, spool.color.g, spool.color.b];
        let mut total_gain = 0.0f32;
        let len = pixel_indices.len();

        match self.mix_model {
            OpticalMixModelType::PorterDuffOcclusion => {
                let s0 = spool_arr[0];
                let s1 = spool_arr[1];
                let s2 = spool_arr[2];
                let is_linear_rgb = self.error_metric == ColorErrorMetric::LinearRgb;

                for i in 0..len {
                    let p = pixel_indices[i] as usize;
                    let imp = state.importance_w[p];
                    if imp <= 0.0 {
                        continue;
                    }
                    let w_sub = (subpixel_weights_u8[i] as f32) * (1.0 / 255.0);
                    let aw = (alpha_eff * w_sub).clamp(0.0, 1.0);
                    if aw <= 1e-7 {
                        continue;
                    }

                    let c0 = state.current_c0[p];
                    let c1 = state.current_c1[p];
                    let c2 = state.current_c2[p];
                    let t0 = state.target_c0[p];
                    let t1 = state.target_c1[p];
                    let t2 = state.target_c2[p];

                    let e0 = c0 - t0;
                    let e1 = c1 - t1;
                    let e2 = c2 - t2;
                    let d0 = c0 - s0;
                    let d1 = c1 - s1;
                    let d2 = c2 - s2;

                    let (dot_ed, norm_d_sq) = if is_linear_rgb {
                        (e0 * d0 + e1 * d1 + e2 * d2, d0 * d0 + d1 * d1 + d2 * d2)
                    } else {
                        let weights = MetricWeights::for_pixel(self.error_metric, c0, t0);
                        (weights.dot([e0, e1, e2], [d0, d1, d2]), weights.norm_sq([d0, d1, d2]))
                    };

                    let raw_gain = 2.0 * aw * dot_ed - (aw * aw) * norm_d_sq;
                    total_gain += imp * raw_gain;
                }
            }
            _ => {
                for i in 0..len {
                    let p = pixel_indices[i] as usize;
                    let w_sub = (subpixel_weights_u8[i] as f32) * (1.0 / 255.0);
                    let imp = state.importance_w[p];
                    if imp <= 0.0 {
                        continue;
                    }

                    let current = [
                        state.current_c0[p],
                        state.current_c1[p],
                        state.current_c2[p],
                    ];
                    let target = [
                        state.target_c0[p],
                        state.target_c1[p],
                        state.target_c2[p],
                    ];

                    let gain = self.pixel_marginal_gain(
                        current,
                        target,
                        spool_arr,
                        alpha_eff,
                        w_sub,
                        imp,
                    );
                    total_gain += gain;
                }
            }
        }

        total_gain
    }

    /// Aplica a corda rasterizada sobre a tela `CanvasStateSoA`, atualizando os pixels in-place.
    pub fn apply_chord(
        &self,
        state: &mut CanvasStateSoA,
        pixel_indices: &[u32],
        subpixel_weights_u8: &[u8],
        spool: &SpoolColor,
        alpha_eff: f32,
    ) {
        let len = pixel_indices.len();
        for i in 0..len {
            let p = pixel_indices[i] as usize;
            let w_sub = (subpixel_weights_u8[i] as f32) * (1.0 / 255.0);
            let cur = state.current_color(p);
            let updated = self.update_pixel_color(cur, spool.color, alpha_eff, w_sub);
            state.set_current_color(p, updated);
        }
    }
}

//! As 5 EstratÃ©gias de MinimizaÃ§Ã£o de Trocas de Carretel (`spool_scheduler.rs`).

//! Na tecelagem fÃ­sica real de String Art, cada troca de carretel exige corte manual da linha,

//! 1. `FreeGreedy`: $\lambda_{\text{troca}} = 0$ (avaliaÃ§Ã£o livre sem penalidades).
//! 2. `HysteresisPenalty`: Aplica penalidade $-\lambda_{\text{troca}}$ ao ganho marginal quando $k \neq k_{\text{atual}}$.
//! 3. `MinRunLength`: Trava a avaliaÃ§Ã£o no carretel ativo atÃ© cumprir uma cota mÃ­nima contÃ­gua $L_{\min}$ de fios.
//! 4. `CoarseToFineEpochs`: Divide os $K_{\max}$ passos em 3 Ã©pocas estruturadas (Base, Refino e Acabamento Topo K/W).

/// As 5 estratÃ©gias de troca de carretel suportadas.
#[derive(Debug, Clone, PartialEq)]

    /// AvaliaÃ§Ã£o livre gulosa sem penalidade de troca ($\lambda = 0$).
    FreeGreedy,

    /// Penalidade de histerese dinÃ¢mica subtraÃ­da do ganho marginal se $k \neq k_{\text{atual}}$.
    HysteresisPenalty {
        penalty_lambda: f32,
    },

    /// Bloqueia a troca enquanto o carretel atual nÃ£o atingir o tamanho mÃ­nimo de lote $L_{\min}$.
    MinRunLength {
        min_run_length: usize,
    },

    /// Ã‰pocas estruturadas Coarse-to-Fine:
    /// - Ã‰poca 1 (Base Volume): Blocos longos de carretÃ©is base cromÃ¡ticos.
    /// - Ã‰poca 2 (Refino CromÃ¡tico): Blocos mÃ©dios de carretÃ©is base.
    /// - Ã‰poca 3 (Acabamento Nitidez no Topo): Ãšltimos ~15% forÃ§am Preto (K) e opcionalmente Branco (W),
    ///   com elevaÃ§Ã£o dinÃ¢mica do peso de bordas no mapa $W(x,y)$.
    CoarseToFineEpochs {
        base_ratio: f32,
        refine_ratio: f32,
        detail_top_ratio: f32,
        black_spool_id: u8,
        white_spool_id: Option<u8>,
        edge_importance_boost: f32,
    },

    /// Camadas sequenciais estritas em lote Ãºnico por cor ($M - 1$ trocas totais no projeto).
    /// Aplica fator de compensaÃ§Ã£o de oclusÃ£o para que camadas inferiores tenham densidade reforÃ§ada.
    SeparatedChannels {
        /// Cotas fracionÃ¡rias de passos para cada carretel (soma deve ser 1.0).
        quotas: Vec<f32>,

/// Estado do escalonador de carretÃ©is ao longo da execuÃ§Ã£o da tecelagem.
#[derive(Debug, Clone, PartialEq)]
pub struct SpoolScheduler {
    pub strategy: SpoolSwitchStrategy,
    pub max_steps: usize,
    pub current_step: usize,
    pub active_spool_id: u8,
    pub consecutive_steps: usize,
    pub total_switches: usize,
}

impl SpoolScheduler {
    /// Cria um novo escalonador de carretÃ©is.
    pub fn new(strategy: SpoolSwitchStrategy, max_steps: usize, initial_spool_id: u8) -> Self {
        Self {
            strategy,
            max_steps: max_steps.max(1),
            current_step: 0,
            active_spool_id: initial_spool_id,
            consecutive_steps: 0,
            total_switches: 0,
        }
    }

    /// Retorna a lista de IDs de carretÃ©is permitidos para avaliaÃ§Ã£o no passo atual.
    pub fn allowed_spools(&self, spools: &[SpoolColor]) -> Vec<u8> {
        if spools.is_empty() {
            return Vec::new();
        }

        let all_ids: Vec<u8> = spools.iter().map(|s| s.id).collect();

        match &self.strategy {
            SpoolSwitchStrategy::FreeGreedy => all_ids,

            SpoolSwitchStrategy::HysteresisPenalty { .. } => all_ids,

            SpoolSwitchStrategy::MinRunLength { min_run_length } => {
                if self.consecutive_steps < *min_run_length && self.current_step > 0 {
                    vec![self.active_spool_id]
                } else {
                    all_ids
                }
            }

            SpoolSwitchStrategy::CoarseToFineEpochs {
                base_ratio,
                refine_ratio,
                black_spool_id,
                white_spool_id,
                ..
            } => {
                let target_spool_id = Self::compute_epoch_spool(
                    self.current_step,
                    self.max_steps,
                    *base_ratio,
                    *refine_ratio,
                    *black_spool_id,
                    *white_spool_id,
                    spools,
                );
                vec![target_spool_id]
            }

            SpoolSwitchStrategy::SeparatedChannels { quotas, .. } => {
                let mut accumulated = 0.0f32;
                let total = self.max_steps as f32;
                let cur = self.current_step as f32;

                for (idx, &quota) in quotas.iter().enumerate() {
                    let next_accum = accumulated + quota * total;
                    if cur < next_accum || idx == quotas.len() - 1 {
                        let spool_id = spools.get(idx).map(|s| s.id).unwrap_or(0);
                        return vec![spool_id];
                    }
                    accumulated = next_accum;
                }
                vec![self.active_spool_id]
            }
        }
    }

    /// Calcula deterministicamente o carretel programado para a Ã©poca atual.
    fn compute_epoch_spool(
        step: usize,
        max_steps: usize,
        base_ratio: f32,
        refine_ratio: f32,
        black_spool_id: u8,
        white_spool_id: Option<u8>,
        spools: &[SpoolColor],
    ) -> u8 {
        let chromatic: Vec<u8> = spools
            .iter()
            .map(|s| s.id)
            .filter(|&id| id != black_spool_id && Some(id) != white_spool_id)
            .collect();

        let progress = (step as f32) / (max_steps as f32);
        let epoch1_limit = base_ratio;
        let epoch2_limit = base_ratio + refine_ratio;

        if progress >= epoch2_limit {
            // Ã‰poca 3 (Acabamento Nitidez no Topo):
            let epoch3_progress = (progress - epoch2_limit) / (1.0 - epoch2_limit).max(1e-4);
            if epoch3_progress < 0.85 || white_spool_id.is_none() {
                black_spool_id
            } else {
                white_spool_id.unwrap()
            }
        } else if chromatic.is_empty() {
            black_spool_id
        } else if progress < epoch1_limit {
            // Ã‰poca 1 (Base Volume): Blocos longos por cor cromÃ¡tica
            let num_c = chromatic.len();
            let step_in_epoch1 = step as f32;
            let epoch1_total = (max_steps as f32) * epoch1_limit;
            let block_idx = ((step_in_epoch1 / epoch1_total.max(1.0)) * (num_c as f32)).floor() as usize;
            let idx = block_idx.min(num_c - 1);
            chromatic[idx]
        } else {
            // Ã‰poca 2 (Refino CromÃ¡tico): Blocos mÃ©dios
            let num_c = chromatic.len();
            let step_in_epoch2 = (step as f32) - (max_steps as f32) * epoch1_limit;
            let epoch2_total = (max_steps as f32) * refine_ratio;
            let block_idx = ((step_in_epoch2 / epoch2_total.max(1.0)) * (num_c as f32)).floor() as usize;
            let idx = block_idx.min(num_c - 1);
            chromatic[idx]
        }
    }

    /// Retorna a penalidade algÃ©brica de troca a ser subtraÃ­da do ganho marginal para o carretel candidato.
    #[inline]
    pub fn switch_penalty(&self, candidate_spool_id: u8) -> f32 {
        if candidate_spool_id == self.active_spool_id || self.current_step == 0 {
            0.0
        } else {
            match &self.strategy {
                SpoolSwitchStrategy::HysteresisPenalty { penalty_lambda } => *penalty_lambda,
                _ => 0.0,
            }
        }
    }

    /// Verifica se a estratÃ©gia atual requer elevaÃ§Ã£o do mapa de bordas $W(p) \leftarrow W(p) + \Delta\beta \cdot E(p)$.
    ///
    /// Ativo na Ã‰poca 3 de `CoarseToFineEpochs`.
    pub fn active_edge_boost(&self) -> f32 {
        match &self.strategy {
            SpoolSwitchStrategy::CoarseToFineEpochs {
                base_ratio,
                refine_ratio,
                edge_importance_boost,
                ..
            } => {
                let progress = (self.current_step as f32) / (self.max_steps as f32);
                if progress >= (*base_ratio + *refine_ratio) {
                    *edge_importance_boost
                } else {
                    0.0
                }
            }
            _ => 0.0,
        }
    }

    /// Fator de compensaÃ§Ã£o de oclusÃ£o para o carretel ativo na estratÃ©gia `SeparatedChannels`.
    pub fn occlusion_compensation_factor(&self, spool_idx: usize, total_spools: usize) -> f32 {
        match &self.strategy {
            SpoolSwitchStrategy::SeparatedChannels {
                occlusion_compensation,
                ..
            } => {
                if total_spools <= 1 {
                    1.0
                } else {
                    let depth_from_top = (total_spools - 1 - spool_idx) as f32;
                    1.0 + (*occlusion_compensation - 1.0) * (depth_from_top / ((total_spools - 1) as f32))
                }
            }
            _ => 1.0,
        }
    }

    /// Registra a aplicaÃ§Ã£o de um passo de corda com o carretel escolhido.
    pub fn advance_step(&mut self, chosen_spool_id: u8, _spools: &[SpoolColor]) {
        if chosen_spool_id != self.active_spool_id {
            self.total_switches += 1;
            self.active_spool_id = chosen_spool_id;
            self.consecutive_steps = 1;
        } else {
            self.consecutive_steps += 1;
        }

        self.current_step += 1;
    }
}

                        from_pin: r.from_pin,
                        to_pin: r.to_pin,
                        spool_id: r.spool_id,
                        departure_winding: r.departure_winding,
                        arrival_winding: r.arrival_winding,
                        gain: r.gain,
                        score_normalized: r.score_normalized,
                    }),

                SolverBackend::GlobalFista(s) => s
                    .step(
                        &mut self.canvas,
                        &mut self.tracker,
                        &mut self.scheduler,
                        &self.matrix,
                        &self.evaluator,
                        &self.spools,
                        self.alpha_eff,
                    )
                    .map(|r| WeaveStepRecord {
                        from_pin: r.from_pin,
                        to_pin: r.to_pin,
                        spool_id: r.spool_id,
                        departure_winding: r.departure_winding,
                        arrival_winding: r.arrival_winding,
                        gain: r.gain,
                        score_normalized: r.score_normalized,
                    }),

                Some(step) => {
                    self.history_steps.push(step.clone());
                    batch_steps.push(step);
                }
                None => {
                    // NÃ£o hÃ¡ mais cordas vÃ¡lidas com ganho positivo

            self.status = SolverStatus::Done;
        }

        self.current_progress(batch_steps)
    }

        }
    }

    pub fn abort(&mut self) {
        self.status = SolverStatus::Aborted;
    }

    /// Verifica se o processo foi concluÃ­do ou abortado.

        self.status == SolverStatus::Done || self.status == SolverStatus::Aborted
    }

//! Algoritmo Guloso Iterativo de String Art (`solvers/greedy.rs`).
//!
//! A partir do pino atual $p_{\text{atual}}$ e para cada cor permitida pelo `SpoolScheduler`,
//! avalia todos os pinos candidatos $p_{\text{prox}}$ lendo fatias contÃ­guas da `SparseChordMatrixCSR`.
//!
//! Normaliza o score pelo comprimento da corda:
//! $$\text{Score}(L, k) = \frac{1}{|L|^\gamma} \sum_{p \in L} \text{Ganho}(p, k) - \text{PenalidadeTroca}(k)$$
//!
//! Garante desempate estritamente determinÃ­stico pelo menor Ã­ndice de pino e menor Ã­ndice de carretel (RNF-02).
//! MantÃ©m histÃ³rico recente (Tabu list) e controle de repetiÃ§Ãµes mÃ¡ximas por par de pinos.

/// Motor de otimizaÃ§Ã£o Iterative Greedy com zero alocaÃ§Ãµes na heap no laÃ§o interno.
pub struct GreedySolver {
    pub config: GreedyConfig,
    pub current_pin: usize,
    pub step_count: usize,
    chord_usage: Vec<u16>,
    tabu_queue: Vec<usize>,
    old_colors_buf: Vec<LinearRgbColor>,
}

impl GreedySolver {
    /// Inicializa um novo solver guloso com buffers prÃ©-alocados para a matriz CSR dada.
    pub fn new(config: GreedyConfig, chord_count: usize) -> Self {
        let current_pin = config.start_pin;
        let tabu_cap = config.tabu_history_size;
        Self {
            config,
            current_pin,
            step_count: 0,
            chord_usage: vec![0; chord_count],
            tabu_queue: Vec::with_capacity(tabu_cap),
            old_colors_buf: Vec::with_capacity(1024),
        }
    }

    /// Reinicia o solver mantendo a memÃ³ria alocada nos buffers.
    pub fn reset(&mut self, start_pin: usize) {
        self.current_pin = start_pin;
        self.step_count = 0;
        self.chord_usage.fill(0);
        self.tabu_queue.clear();
        self.old_colors_buf.clear();
    }

    /// Retorna o nÃºmero de vezes que a corda de Ã­ndice canÃ´nico $k$ foi utilizada.
    #[inline]
    pub fn chord_usage(&self, chord_idx: usize) -> u16 {
        self.chord_usage.get(chord_idx).copied().unwrap_or(0)
    }

    /// Retorna uma referÃªncia imutÃ¡vel a todo o vetor de contagem de uso de cordas.
    #[inline]
    pub fn all_chord_usages(&self) -> &[u16] {
        &self.chord_usage
    }

    /// Verifica se um pino estÃ¡ na lista Tabu recente.
    #[inline]
    pub fn is_tabu(&self, pin: usize) -> bool {
        self.tabu_queue.contains(&pin)
    }

    /// Registra um pino na fila Tabu FIFO.
    fn push_tabu(&mut self, pin: usize) {
        if self.config.tabu_history_size == 0 {
            return;
        }
        if self.tabu_queue.len() >= self.config.tabu_history_size {
            self.tabu_queue.remove(0);
        }
        self.tabu_queue.push(pin);
    }

    /// Avalia todos os pinos candidatos $p_{\text{prox}}$ a partir de `current_pin`
    /// para cada carretel permitido pelo `SpoolScheduler` e seleciona o melhor acorde.
    ///
    /// Aplica desempate estritamente determinÃ­stico:
    /// - Maior score;
    /// - Em caso de empate: menor Ã­ndice de pino `to_pin`;
    /// - Em caso de novo empate: menor Ã­ndice `spool_id`.
    pub fn find_best_chord(
        &self,
        canvas: &CanvasStateSoA,
        matrix: &SparseChordMatrixCSR,
        evaluator: &OpticalModelEvaluator,
        scheduler: &SpoolScheduler,
        spools: &[SpoolColor],
        alpha_eff: f32,
    ) -> Option<GreedyChordCandidate> {
        let pin_count = matrix.pin_count;
        if pin_count < 2 {
            return None;
        }

        let allowed_spool_ids = scheduler.allowed_spools(spools);
        if allowed_spool_ids.is_empty() {
            return None;
        }

        let mut best: Option<GreedyChordCandidate> = None;
        let gamma = self.config.length_exponent_gamma;
        let max_repeats = self.config.max_chord_repeats;

        for next_pin in 0..pin_count {
            if next_pin == self.current_pin {
                continue;
            }

            if self.is_tabu(next_pin) {
                continue;
            }

            let chord_idx = match SparseChordMatrixCSR::try_canonical_chord_index(
                self.current_pin,
                next_pin,
                pin_count,
            ) {
                Some(idx) => idx,
                None => continue,
            };

            if (self.chord_usage[chord_idx] as usize) >= max_repeats {
                continue;
            }

            let (pixel_indices, subpixel_weights) = matrix.chord_slice(chord_idx);
            if pixel_indices.is_empty() {
                continue;
            }

            let chord_len = pixel_indices.len() as f32;
            let len_factor = if gamma == 1.0 {
                chord_len
            } else if gamma == 0.0 {
                1.0
            } else {
                chord_len.powf(gamma).max(1e-5)
            };

            for &spool_id in &allowed_spool_ids {
                let spool = match spools.iter().find(|s| s.id == spool_id) {
                    Some(s) => s,
                    None => continue,
                };

                let raw_gain = evaluator.chord_marginal_gain(
                    canvas,
                    pixel_indices,
                    subpixel_weights,
                    spool,
                    alpha_eff,
                );

                // Apenas cordas que efetivamente reduzem o erro Ã³ptico sÃ£o candidatas vÃ¡lidas
                if raw_gain <= 0.0 {
                    continue;
                }

                let penalty = scheduler.switch_penalty(spool.id);
                let score = (raw_gain / len_factor) - penalty;

                if score <= 0.0 {
                    continue;
                }

                let is_better = match &best {
                    None => true,
                    Some(b) => {
                        if score > b.score {
                            true
                        } else if (score - b.score).abs() <= 1e-7 {
                            // Desempate determinÃ­stico (RNF-02)
                            if next_pin < b.to_pin {
                                true
                            } else if next_pin == b.to_pin {
                                spool.id < b.spool_id
                            } else {
                                false
                            }
                        } else {
                            false
                        }
                    }
                };

                if is_better {
                    best = Some(GreedyChordCandidate {
                        from_pin: self.current_pin,
                        to_pin: next_pin,
                        spool_id: spool.id,
                        chord_idx,
                        raw_gain,
                        score,
                    });
                }
            }
        }

        best
    }

    /// Executa um Ãºnico passo de tecelagem gulosa:
    /// 1. Seleciona a melhor corda candidata determinÃ­stica;
    /// 2. Salva as cores antigas no buffer reusÃ¡vel sem alocaÃ§Ãµes;
    /// 3. Aplica a corda sobre `canvas`;
    /// 4. Atualiza incrementalmente o `tracker` em tempo $O(|L|)$;
    /// 5. AvanÃ§a o `scheduler` e aplica reforÃ§o dinÃ¢mico de bordas se necessÃ¡rio;
    /// 6. Atualiza uso de cordas, pino atual e fila tabu;
    /// 7. Retorna o registro do passo.
    pub fn step(
        &mut self,
        canvas: &mut CanvasStateSoA,
        tracker: &mut IncrementalMetricsTracker,
        scheduler: &mut SpoolScheduler,
        matrix: &SparseChordMatrixCSR,
        evaluator: &OpticalModelEvaluator,
        spools: &[SpoolColor],
        alpha_eff: f32,
    ) -> Option<GreedyStepRecord> {
        let candidate = self.find_best_chord(canvas, matrix, evaluator, scheduler, spools, alpha_eff)?;

        let (pixel_indices, subpixel_weights) = matrix.chord_slice(candidate.chord_idx);
        let spool = spools
            .iter()
            .find(|s| s.id == candidate.spool_id)
            .expect("Spool must exist");

        // 1. Coleta cores antigas para atualizaÃ§Ã£o incremental de telemetria O(|L|)
        self.old_colors_buf.clear();
        for &p in pixel_indices {
            self.old_colors_buf.push(canvas.current_color(p as usize));
        }

        // 2. Aplica a corda in-place no CanvasStateSoA
        evaluator.apply_chord(canvas, pixel_indices, subpixel_weights, spool, alpha_eff);

        // 3. Notifica o tracker incremental
        tracker.update_pixels(canvas, pixel_indices, &self.old_colors_buf);

        // 4. AvanÃ§a o agendador de carretÃ©is
        scheduler.advance_step(spool.id, spools);
        let edge_boost = scheduler.active_edge_boost();
        if edge_boost > 0.0 {
            canvas.apply_edge_importance_boost(edge_boost);
        }

        // 5. Atualiza estado do solver
        self.chord_usage[candidate.chord_idx] += 1;
        let from = self.current_pin;
        let to = candidate.to_pin;
        self.push_tabu(from);
        self.current_pin = to;
        self.step_count += 1;

        let score_norm = ((candidate.score.max(0.0) / (candidate.score.max(0.0) + 1.0)) * 65535.0)
            .clamp(0.0, 65535.0) as u16;

        Some(GreedyStepRecord {
            from_pin: from as u16,
            to_pin: to as u16,
            spool_id: candidate.spool_id,
            departure_winding: PegSide::Clockwise,
            arrival_winding: PegSide::Clockwise,
            gain: candidate.raw_gain,
            score_normalized: score_norm,
        })
    }
}

/// Motor de tecelagem exposto para o WebAssembly e Web Worker via wasm-bindgen.
#[wasm_bindgen]
pub struct WasmWeavingEngine {
    is_shared: bool,
    capacity_steps: usize,

    control_int32: js_sys::Int32Array,
    control_float32: js_sys::Float32Array,
    path_uint8: js_sys::Uint8Array,

    dsp_pipeline: DspPipeline,
    loom: Option<LoomGeometry>,
    last_source_rgba: Option<Rgba8Image>,
    base_importance_weights: Vec<f32>,
    current_importance_weights: Vec<f32>,

    cached_matrix: Option<(LoomPhysicalSpec, Pixels, u16, RasterizerType, Arc<SparseChordMatrixCSR>)>,
    solver_session: Option<SolverSession>,
    spool_infos: Vec<SpoolInfo>,

    diagnostic_prepared_rgba: Vec<u8>,
    diagnostic_importance_rgba: Vec<u8>,
    diagnostic_edges_rgba: Vec<u8>,
    diagnostic_radon_rgba: Vec<u8>,

    total_thread_meters: f32,
    spool_thread_meters: [f32; 16],

#[wasm_bindgen]
impl WasmWeavingEngine {
    /// Inicializa a ponte Wasm conectada ao `SharedArrayBuffer` (ou `ArrayBuffer` de fallback).
    #[wasm_bindgen(constructor)]
    pub fn new(shared_buffer: JsValue) -> Result<WasmWeavingEngine, JsValue> {
        let byte_length = js_sys::Reflect::get(&shared_buffer, &JsValue::from_str("byteLength"))
            .map_err(|_| JsValue::from_str("Invalid buffer: missing byteLength"))?
            .as_f64()
            .ok_or_else(|| JsValue::from_str("byteLength is not a number"))? as usize;

        if byte_length < 128 {
            return Err(JsValue::from_str("Buffer is smaller than 128-byte ControlBlock"));
        }

        let capacity_steps = (byte_length - 128) / 8;

        let is_shared = js_sys::Reflect::get(&shared_buffer, &JsValue::from_str("constructor"))
            .ok()
            .and_then(|c| js_sys::Reflect::get(&c, &JsValue::from_str("name")).ok())
            .and_then(|n| n.as_string())
            .map(|name| name == "SharedArrayBuffer")
            .unwrap_or(false);

        let control_int32 = js_sys::Int32Array::new_with_byte_offset_and_length(&shared_buffer, 0, 8);
        let control_float32 = js_sys::Float32Array::new_with_byte_offset_and_length(&shared_buffer, 32, 24);
        let path_uint8 = js_sys::Uint8Array::new_with_byte_offset_and_length(
            &shared_buffer,
            128,
            (capacity_steps * 8) as u32,
        );

        let dsp_config = DspPipelineConfig::default();
        let dsp_pipeline = DspPipeline::new(dsp_config);

        Ok(Self {
            is_shared,
            capacity_steps,
            control_int32,
            control_float32,
            path_uint8,
            dsp_pipeline,
            loom: None,
            last_source_rgba: None,
            base_importance_weights: Vec::new(),
            current_importance_weights: Vec::new(),
            cached_matrix: None,
            solver_session: None,
            spool_infos: Vec::new(),
            diagnostic_prepared_rgba: Vec::new(),
            diagnostic_importance_rgba: Vec::new(),
            diagnostic_edges_rgba: Vec::new(),
            diagnostic_radon_rgba: Vec::new(),
            diagnostic_dimensions: [0; 8],
            total_thread_meters: 0.0,
            spool_thread_meters: [0.0; 16],
        })
    }

    /// Executa o pipeline `weaver-dsp` completo e preenche os 4 buffers de diagnÃ³stico RGBA.
    pub fn load_and_prepare_image(
        &mut self,
        raw_rgba: &[u8],
        width: u32,
        height: u32,
        config_json: &str,
    ) -> Result<String, JsValue> {
        let config: WasmWeaverConfig = if config_json.trim().is_empty() {
            WasmWeaverConfig::default()
        } else {
            serde_json::from_str(config_json)
                .map_err(|e| JsValue::from_str(&format!("JSON deserialize error: {}", e)))?
        };

        let loom = build_loom_from_config(&config);
        let res = config.target_resolution;

        let mut dsp_config = DspPipelineConfig::default();
        dsp_config.target_resolution = res;
        dsp_config.beta_face = config.face_importance_weight;
        dsp_config.beta_edge = config.edge_blend_weight;
        dsp_config.high_freq_blend = config.edge_blend_weight.min(0.5);

        if config.clahe_clip_limit > 0.0 {
            dsp_config.clahe = Some(ClaheConfig {
                clip_limit: config.clahe_clip_limit,
                ..Default::default()
            });
        } else {
            dsp_config.clahe = None;
        }

        if let Some(f) = &config.framing {
            dsp_config.framing = FramingConfig {
                zoom: f.zoom,
                pan_x: f.pan_x_percent,
                pan_y: f.pan_y_percent,
                rotation_deg: f.rotation_deg,
            };
            if f.auto_face_target_ratio > 0.0 {
                dsp_config.auto_framing = Some(AutoFramingOptions {
                    target_ratio: f.auto_face_target_ratio,
                    ..Default::default()
                });
            }
        }

        self.dsp_pipeline = DspPipeline::new(dsp_config);
        let src_img = Rgba8Image::new(width, height, 4, raw_rgba.to_vec());

        let diag = self.dsp_pipeline.execute(&src_img, Some(&loom), None);

        self.diagnostic_prepared_rgba = diag.prepared_target_rgba.data.clone();
        self.diagnostic_importance_rgba = diag.importance_map_rgba.data.clone();
        self.diagnostic_edges_rgba = diag.edges_rgba.data.clone();
        self.diagnostic_radon_rgba = diag.radon_sinogram_rgba.data.clone();

        self.diagnostic_dimensions = [
            diag.prepared_target_rgba.width,
            diag.prepared_target_rgba.height,
            diag.importance_map_rgba.width,
            diag.importance_map_rgba.height,
            diag.edges_rgba.width,
            diag.edges_rgba.height,
            diag.radon_sinogram_rgba.width,
            diag.radon_sinogram_rgba.height,
        ];

        let base_imp = self.dsp_pipeline.importance_map().unwrap().weights.clone();
        self.base_importance_weights = base_imp.clone();
        self.current_importance_weights = base_imp;

        self.loom = Some(loom);
        self.last_source_rgba = Some(src_img);

        let summary = format!(
            r#"{{"preparedWidth":{},"preparedHeight":{},"importanceWidth":{},"importanceHeight":{},"edgesWidth":{},"edgesHeight":{},"radonWidth":{},"radonHeight":{}}}"#,
            self.diagnostic_dimensions[0],
            self.diagnostic_dimensions[1],
            self.diagnostic_dimensions[2],
            self.diagnostic_dimensions[3],
            self.diagnostic_dimensions[4],
            self.diagnostic_dimensions[5],
            self.diagnostic_dimensions[6],
            self.diagnostic_dimensions[7]
        );

        Ok(summary)
    }

    /// Atualiza a mÃ¡scara manual de foco facial sem recalcular os estÃ¡gios anteriores do DSP.
    pub fn apply_user_importance_brush(&mut self, brush_weights: &[f32]) {
        if self.base_importance_weights.is_empty()
            || self.base_importance_weights.len() != brush_weights.len()
        {
            return;
        }

        let mut updated = self.base_importance_weights.clone();
        for i in 0..updated.len() {
            updated[i] = (updated[i] + brush_weights[i]).max(0.0);
        }

        let res = self.diagnostic_dimensions[2]; // importance width
        let imp_map = ImportanceMap::new(res, res, updated.clone());
        self.diagnostic_importance_rgba = imp_map.to_preview_rgba().data;
        self.current_importance_weights = updated;

        if let Some(session) = &mut self.solver_session {
            for (i, &w) in self.current_importance_weights.iter().enumerate() {
                if i < session.canvas.importance_w.len() {
                    session.canvas.importance_w[i] = w;
                }
            }
        }
    }

    /// ConstrÃ³i ou reutiliza a `SparseChordMatrixCSR` e inicializa o motor do solver.
    pub fn init_solver(&mut self, config_json: &str) -> Result<(), JsValue> {
        let config: WasmWeaverConfig = if config_json.trim().is_empty() {
            WasmWeaverConfig::default()
        } else {
            serde_json::from_str(config_json)
                .map_err(|e| JsValue::from_str(&format!("JSON deserialize error: {}", e)))?
        };

        let loom = self
            .loom
            .as_ref()
            .cloned()
            .unwrap_or_else(|| build_loom_from_config(&config));
        let res = config.target_resolution;

        let rasterizer = match config.rasterizer.to_uppercase().as_str() {
            "BRESENHAM_BINARY" => RasterizerType::BresenhamBinary,
            _ => RasterizerType::XiaolinWuAntialiased,
        };

        let min_gap = 3u16;

        let matrix: Arc<SparseChordMatrixCSR> = match &self.cached_matrix {
            Some((spec, r, gap, rast, m))
                if *spec == loom.spec && r.0 == res && *gap == min_gap && *rast == rasterizer =>
            {
                m.clone()
            }
            _ => {
                let m = Arc::new(SparseChordMatrixCSR::build(
                    &loom,
                    Pixels(res),
                    min_gap as usize,
                    rasterizer,
                ));
                self.cached_matrix =
                    Some((loom.spec.clone(), Pixels(res), min_gap, rasterizer, m.clone()));
                m
            }
        };

        let prepared_gray = self
            .dsp_pipeline
            .prepared_gray()
            .cloned()
            .unwrap_or_else(|| weaver_dsp::GrayImage::fill(res, res, 1, 0.5));

        let target_colors = prepared_gray.to_linear_rgb_vec();
        let importance_slice = if !self.current_importance_weights.is_empty()
            && self.current_importance_weights.len() == (res * res) as usize
        {
            Some(self.current_importance_weights.as_slice())
        } else {
            None
        };

        let canvas = CanvasStateSoA::from_target_and_background(
            res,
            res,
            &target_colors,
            LinearRgbColor::WHITE,
            importance_slice,
        );

        let mix_model = match config.optical_mix_model.to_uppercase().as_str() {
            "LINEAR_SUBTRACTIVE" => OpticalMixModelType::LinearSubtractive,
            "BEER_LAMBERT" => OpticalMixModelType::BeerLambertTransmittance,
            _ => OpticalMixModelType::PorterDuffOcclusion,
        };

        let err_metric = match config.error_metric.to_uppercase().as_str() {
            "REDMEAN_WEIGHTED" => ColorErrorMetric::RedmeanWeighted,
            "CIELAB_DELTA_E2000" => ColorErrorMetric::CielabDeltaE2000,
            _ => ColorErrorMetric::LinearRgb,
        };

        let evaluator = OpticalModelEvaluator::new(mix_model, err_metric);

        let (spools, spool_infos) = build_spools(&config);
        self.spool_infos = spool_infos;

        let strat = match config.spool_strategy.to_uppercase().as_str() {
            "HYSTERESIS_PENALTY" => SpoolSwitchStrategy::HysteresisPenalty {
                penalty_lambda: if config.switch_penalty_lambda > 0.0 {
                    config.switch_penalty_lambda
                } else {
                    0.05
                },
            },
            "MIN_RUN_LENGTH" => SpoolSwitchStrategy::MinRunLength {
                min_run_length: config.min_run_length.max(10),
            },
            "COARSE_TO_FINE_EPOCHS" => SpoolSwitchStrategy::CoarseToFineEpochs {
                base_ratio: 0.50,
                refine_ratio: 0.35,
                detail_top_ratio: 0.15,
                black_spool_id: if spools.len() > 1 {
                    (spools.len() - 1) as u8
                } else {
                    0
                },
                white_spool_id: None,
                edge_importance_boost: 2.0,
            },
            "SEPARATED_CHANNELS" => {
                let n = spools.len();
                let q = 1.0 / n as f32;
                SpoolSwitchStrategy::SeparatedChannels {
                    quotas: vec![q; n],
                    occlusion_compensation: 1.2,
                }
            }
            _ => SpoolSwitchStrategy::FreeGreedy,
        };

        let scheduler = SpoolScheduler::new(strat, config.max_lines, 0);

        let alpha_eff = compute_effective_alpha(
            0.30,
            Millimeters(config.thread_thickness_mm),
            Pixels(res),
            Millimeters(config.diameter_mm as f32),
        );

        let session = match config.solver.to_uppercase().as_str() {
            "BEAM_SEARCH" => {
                let b_cfg = BeamSearchConfig {
                    max_steps: config.max_lines,
                    beam_width: 8,
                    lookahead_depth: 3,
                    max_chord_repeats: config.max_chord_repeats.max(1),
                    length_exponent_gamma: 1.0,
                    start_pin: 0,
                };
                SolverSession::new_beam_search(
                    b_cfg, canvas, matrix, evaluator, scheduler, spools, alpha_eff,
                )
            }
            "GLOBAL_NNLS_FISTA" => {
                let f_cfg = GlobalFistaConfig {
                    max_steps: config.max_lines,
                    fista_iterations: 40,
                    l1_regularization_lambda: 0.01,
                    painter_layers_per_color: 1,
                    max_chord_repeats: config.max_chord_repeats.max(1),
                    start_pin: 0,
                };
                SolverSession::new_global_fista(
                    f_cfg, canvas, matrix, evaluator, scheduler, spools, alpha_eff,
                )
            }
            _ => {
                let g_cfg = GreedyConfig {
                    max_steps: config.max_lines,
                    max_chord_repeats: config.max_chord_repeats.max(1),
                    length_exponent_gamma: 1.0,
                    start_pin: 0,
                    tabu_history_size: 0,
                };
                SolverSession::new_greedy(
                    g_cfg, canvas, matrix, evaluator, scheduler, spools, alpha_eff,
                )
            }
        };

        self.solver_session = Some(session);
        self.loom = Some(loom);
        self.total_thread_meters = 0.0;
        self.spool_thread_meters = [0.0; 16];

        self.write_status(SolverStatus::Idle);
        self.write_step_index(0);
        self.write_active_spool(0);
        self.write_total_switches(0);

        Ok(())
    }

    /// Executa um lote de passos (ex: 50), escreve diretamente no `SharedArrayBuffer` e atualiza telemetria.
    /// Retorna `false` quando concluÃ­do ou interrompido.
    pub fn step_solver_batch(&mut self, batch_size: u32) -> bool {
        if self.solver_session.is_none() {
            return false;
        }

        // Verifica interrupÃ§Ã£o pelo ControlBlock (PAUSE=2, ABORT=4)
        let current_status_code = self.read_status_code();
        if current_status_code == 4 {
            if let Some(session) = &mut self.solver_session {
                session.abort();
            }
            self.write_status(SolverStatus::Aborted);
            return false;
        } else if current_status_code == 2 {
            if let Some(session) = &mut self.solver_session {
                session.pause();
            }
            self.write_status(SolverStatus::Paused);
            return false;
        }

        let session = self.solver_session.as_mut().unwrap();
        let prev_step = session.history_steps.len();
        let progress = session.step_batch(batch_size as usize);
        let batch_len = progress.batch_steps.len();

        let loom_ref = self.loom.as_ref();
        let pin_d = loom_ref.map(|l| l.spec.pin_diameter_mm).unwrap_or(2.0);

        // Escreve cada passo diretamente no WeavingPathBuffer (byte 128 em diante)
        if batch_len > 0 {
            let mut packed_bytes = Vec::with_capacity(batch_len * 8);

            for (idx, step) in progress.batch_steps.iter().enumerate() {
                let step_idx = prev_step + idx;
                if step_idx >= self.capacity_steps {
                    break;
                }

                let from_bytes = step.from_pin.to_le_bytes();
                let to_bytes = step.to_pin.to_le_bytes();
                let spool_byte = step.spool_id;
                let dep_flag = if step.departure_winding == PegSide::CounterClockwise {
                    1u8
                } else {
                    0u8
                };
                let arr_flag = if step.arrival_winding == PegSide::CounterClockwise {
                    2u8
                } else {
                    0u8
                };
                let flags = dep_flag | arr_flag;
                let score_bytes = step.score_normalized.to_le_bytes();

                packed_bytes.push(from_bytes[0]);
                packed_bytes.push(from_bytes[1]);
                packed_bytes.push(to_bytes[0]);
                packed_bytes.push(to_bytes[1]);
                packed_bytes.push(spool_byte);
                packed_bytes.push(flags);
                packed_bytes.push(score_bytes[0]);
                packed_bytes.push(score_bytes[1]);

                // Acumula metragem do fio
                let chord_len_mm = if let Some(loom) = loom_ref {
                    let p_a = loom.pin_position(step.from_pin as usize);
                    let p_b = loom.pin_position(step.to_pin as usize);
                    let dx = p_a.0 - p_b.0;
                    let dy = p_a.1 - p_b.1;
                    (dx * dx + dy * dy).sqrt() + std::f64::consts::PI * pin_d
                } else {
                    500.0
                };
                let chord_meters = (chord_len_mm / 1000.0) as f32;
                self.total_thread_meters += chord_meters;
                let s_id = step.spool_id as usize;
                if s_id < 16 {
                    self.spool_thread_meters[s_id] += chord_meters;
                }
            }

            let start_byte_offset = (prev_step * 8) as u32;
            let end_byte_offset = start_byte_offset + packed_bytes.len() as u32;
            self.path_uint8
                .subarray(start_byte_offset, end_byte_offset)
                .copy_from(&packed_bytes);
        }

        // Atualiza ControlBlock atomicamente
        self.write_step_index(progress.current_step as i32);
        self.write_active_spool(progress.active_spool_id as i32);
        self.write_total_switches(progress.total_switches as i32);
        self.write_status(progress.status);

        self.control_float32
            .set_index(CTRL_FLOAT32_RMSE as u32, progress.telemetry.rmse);
        self.control_float32
            .set_index(CTRL_FLOAT32_DELTA_E as u32, progress.telemetry.delta_e);
        self.control_float32
            .set_index(CTRL_FLOAT32_SSIM as u32, progress.telemetry.ssim);
        self.control_float32
            .set_index(CTRL_FLOAT32_TOTAL_METERS as u32, self.total_thread_meters);

        for (i, &meters) in self.spool_thread_meters.iter().enumerate().take(16) {
            self.control_float32
                .set_index((CTRL_FLOAT32_SPOOL_METERS_BASE + i) as u32, meters);
        }

        progress.status == SolverStatus::Running
    }

    // --- Helpers Internos de Atomics / ArrayBuffer ---

    fn read_status_code(&self) -> i32 {
        if self.is_shared {
            js_sys::Atomics::load(&self.control_int32, CTRL_INT32_STATUS_FLAG as u32).unwrap_or(0)
        } else {
            self.control_int32.get_index(CTRL_INT32_STATUS_FLAG as u32)
        }
    }

    fn write_status(&self, status: SolverStatus) {
        let val = status.as_i32();
        if self.is_shared {
            let _ = js_sys::Atomics::store(&self.control_int32, CTRL_INT32_STATUS_FLAG as u32, val);
        } else {
            self.control_int32
                .set_index(CTRL_INT32_STATUS_FLAG as u32, val);
        }
    }

    fn write_step_index(&self, step: i32) {
        if self.is_shared {
            let _ = js_sys::Atomics::store(
                &self.control_int32,
                CTRL_INT32_CURRENT_STEP as u32,
                step,
            );
        } else {
            self.control_int32
                .set_index(CTRL_INT32_CURRENT_STEP as u32, step);
        }
    }

    fn write_active_spool(&self, spool: i32) {
        if self.is_shared {
            let _ = js_sys::Atomics::store(
                &self.control_int32,
                CTRL_INT32_ACTIVE_SPOOL as u32,
                spool,
            );
        } else {
            self.control_int32
                .set_index(CTRL_INT32_ACTIVE_SPOOL as u32, spool);
        }
    }

    fn write_total_switches(&self, switches: i32) {
        if self.is_shared {
            let _ = js_sys::Atomics::store(
                &self.control_int32,
                CTRL_INT32_TOTAL_SWITCHES as u32,
                switches,
            );
        } else {
            self.control_int32
                .set_index(CTRL_INT32_TOTAL_SWITCHES as u32, switches);
        }
    }

    fn collect_weaving_steps(&self) -> Vec<WeavingStep> {
        let session = match &self.solver_session {
            Some(s) => s,
            None => return Vec::new(),
        };

        session
            .history_steps
            .iter()
            .enumerate()
            .map(|(i, r)| {
                WeavingStep::with_details(
                    i,
                    r.from_pin,
                    r.to_pin,
                    r.spool_id,
                    match r.departure_winding {
                        PegSide::Clockwise => WindingDirection::Clockwise,
                        PegSide::CounterClockwise => WindingDirection::CounterClockwise,
                    },
                    match r.arrival_winding {
                        PegSide::Clockwise => WindingDirection::Clockwise,
                        PegSide::CounterClockwise => WindingDirection::CounterClockwise,
                    },
                    r.score_normalized,
                    r.spool_id,
                )
            })
            .collect()
    }
}

fn build_loom_from_config(cfg: &WasmWeaverConfig) -> LoomGeometry {
    let shape = cfg.shape.to_uppercase();
    let spec = if shape == "ELLIPTICAL" {
        LoomPhysicalSpec::ellipse(
            cfg.width_mm,
            cfg.height_mm,
            cfg.pin_count,
            cfg.pin_diameter_mm,
            cfg.drill_bit_mm,
            cfg.edge_margin_mm,
        )
    } else if shape == "RECTANGULAR" {
        LoomPhysicalSpec::rectangle(
            cfg.width_mm,
            cfg.height_mm,
            cfg.pin_count,
            cfg.pin_diameter_mm,
            cfg.drill_bit_mm,
            cfg.edge_margin_mm,
        )
    } else if shape == "REGULAR_POLYGON" {
        LoomPhysicalSpec::regular_polygon(
            cfg.polygon_sides,
            cfg.diameter_mm,
            cfg.pin_count,
            cfg.pin_diameter_mm,
            cfg.drill_bit_mm,
            cfg.edge_margin_mm,
        )
    } else {
        LoomPhysicalSpec::circle(
            cfg.diameter_mm,
            cfg.pin_count,
            cfg.pin_diameter_mm,
            cfg.drill_bit_mm,
            cfg.edge_margin_mm,
        )
    };
    LoomGeometry::new(spec)
}

fn build_spools(cfg: &WasmWeaverConfig) -> (Vec<SpoolColor>, Vec<SpoolInfo>) {
    match cfg.palette_mode.to_uppercase().as_str() {
        "CMYK_SUBTRACTIVE" => {
            let spools = vec![
                SpoolColor::new(0, LinearRgbColor::new(1.0, 1.0, 0.0), 0.25), // Yellow
                SpoolColor::new(1, LinearRgbColor::new(1.0, 0.0, 1.0), 0.25), // Magenta
                SpoolColor::new(2, LinearRgbColor::new(0.0, 1.0, 1.0), 0.25), // Cyan
                SpoolColor::new(3, LinearRgbColor::BLACK, 0.35),               // Black
            ];
            let infos = vec![
                SpoolInfo::new(0, "Amarelo", "#ffff00"),
                SpoolInfo::new(1, "Magenta", "#ff00ff"),
                SpoolInfo::new(2, "Ciano", "#00ffff"),
                SpoolInfo::new(3, "Preto", "#000000"),
            ];
            (spools, infos)
        }
        "RGB_OCCLUSIVE" => {
            let spools = vec![
                SpoolColor::new(0, LinearRgbColor::new(1.0, 0.0, 0.0), 0.30),
                SpoolColor::new(1, LinearRgbColor::new(0.0, 1.0, 0.0), 0.30),
                SpoolColor::new(2, LinearRgbColor::new(0.0, 0.0, 1.0), 0.30),
                SpoolColor::new(3, LinearRgbColor::BLACK, 0.35),
            ];
            let infos = vec![
                SpoolInfo::new(0, "Vermelho", "#ff0000"),
                SpoolInfo::new(1, "Verde", "#00ff00"),
                SpoolInfo::new(2, "Azul", "#0000ff"),
                SpoolInfo::new(3, "Preto", "#000000"),
            ];
            (spools, infos)
        }
        _ => {
            let spools = vec![SpoolColor::new(0, LinearRgbColor::BLACK, 0.35)];
            let infos = vec![SpoolInfo::new(0, "Preto", "#000000")];
            (spools, infos)
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn wasm_adapter_reexports_inner_layers() {
        assert_eq!(RING_BUFFER_HEADER_WORDS, 4);
        let grid = DspGridSpec {
            resolution_px: Pixels(500),
        };
        assert_eq!(grid.resolution_px.0, 500);
    }

    #[test]
    fn wasm_config_deserialization_defaults() {
        let json = r#"{"shape":"CIRCULAR","pinCount":200,"diameterMm":600.0}"#;
        let cfg: WasmWeaverConfig = serde_json::from_str(json).unwrap();
        assert_eq!(cfg.pin_count, 200);
        assert_eq!(cfg.diameter_mm, 600.0);
        assert_eq!(cfg.solver, "ITERATIVE_GREEDY");
        assert_eq!(cfg.palette_mode, "MONOCHROME");
        assert_eq!(cfg.target_resolution, 500);
    }

    #[test]
    fn wasm_loom_builder_shapes() {
        let mut cfg = WasmWeaverConfig::default();
        cfg.shape = "ELLIPTICAL".to_string();
        cfg.width_mm = 500.0;
        cfg.height_mm = 400.0;
        cfg.pin_count = 100;
        let loom = build_loom_from_config(&cfg);
        assert_eq!(loom.pins().len(), 100);

        cfg.shape = "RECTANGULAR".to_string();
        let loom_rect = build_loom_from_config(&cfg);
        assert_eq!(loom_rect.pins().len(), 100);
    }

    #[test]
    fn wasm_spool_builder_modes() {
        let mut cfg = WasmWeaverConfig::default();
        cfg.palette_mode = "CMYK_SUBTRACTIVE".to_string();
        let (spools_cmyk, infos_cmyk) = build_spools(&cfg);
        assert_eq!(spools_cmyk.len(), 4);
        assert_eq!(infos_cmyk.len(), 4);

        cfg.palette_mode = "MONOCHROME".to_string();
        let (spools_mono, infos_mono) = build_spools(&cfg);
        assert_eq!(spools_mono.len(), 1);
        assert_eq!(infos_mono.len(), 1);
    }
}
